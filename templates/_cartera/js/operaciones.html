if (sortDirections[sortKey] === undefined) { sortDirections[sortKey] = true; } const ascending = sortDirections[sortKey]; // Filter out rows without enough cells (e.g. empty messages) const validRows = rows.filter(r => r.cells.length > columnIndex); validRows.sort((a, b) => { let valA = a.cells[columnIndex].innerText.trim(); let valB = b.cells[columnIndex].innerText.trim(); if (a.cells[columnIndex].hasAttribute('data-sort-value')) { valA = a.cells[columnIndex].getAttribute('data-sort-value'); if (ga !== gb) return ga - gb; if (isNaN(numA) || isNaN(numB)) return isNaN(numA) ? (isNaN(numB) ? 0 : 1) : -1; return ascending ? numA - numB : numB - numA; } else { return ascending ? valA.localeCompare(valB) : valB.localeCompare(valA); } }); sortDirections[sortKey] = !ascending; validRows.forEach(row => tbody.appendChild(row)); // Re-aplicar filtros/paginado si es una de nuestras tablas paginadas if (tableId === 'closedPositionsTable') { if (typeof filterClosedPositions === 'function') { filterClosedPositions(); } } else if (tableId === 'closedTable') { if (typeof applyHistFilters === 'function') { applyHistFilters(false); // Mantener página actual de operaciones detalladas } } // Actualizar visualmente la cabecera (iconos y clases de Bootstrap) const headers = table.querySelectorAll("thead th"); headers.forEach((th, idx) => { const icon = th.querySelector("i.bi"); if (icon) { if (idx === columnIndex) { // Columna ordenada activa icon.className = ascending ?
