${company ? company.name : ""} ${ticker}
`; }); } } else { const matches = analyzedCompanies.filter(c => c.ticker.toLowerCase().includes(text) || c.name.toLowerCase().includes(text) ).slice(0, 10); if (matches.length > 0) { matches.forEach(m => { html += `
${m.name} ${m.ticker}
`; }); } } if (html) { suggestions.innerHTML = html; suggestions.style.display = 'block'; } else { suggestions.style.display = 'none'; } }; const highlightItem = (index) => { const items = suggestions.querySelectorAll('.suggestion-item'); } } else if (e.key === 'ArrowUp') { e.preventDefault(); if (selectedIndex > 0) { selectedIndex--; highlightItem(selectedIndex); } else if (selectedIndex === -1) { selectedIndex = items.length - 1; highlightItem(selectedIndex); } } else if (e.key === 'Enter') { if (selectedIndex >= 0 && selectedIndex < items.length) { e.preventDefault(); const ticker = items[selectedIndex].getAttribute('data-ticker'); input.value = ticker; suggestions.style.display = 'none'; if (input.form && opts.submit !== false) input.form.submit(); } } else if (e.key === 'Escape') { suggestions.style.display = 'none'; } }); suggestions.addEventListener('mousedown', (e) => { const item = e.target.closest('.suggestion-item');
